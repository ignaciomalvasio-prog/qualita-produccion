/**
 * Puente entre la pestaña Cámara de la app Qualitá y la Google Sheet "FAENA POR DÍA" (hoja "Faena-Desposte").
 * Se instala DENTRO de la planilla: Extensiones → Apps Script → pegar este código → Implementar como aplicación web
 * (ejecutar como: yo; acceso: cualquier persona). La primera vez que la app se conecta recibe una clave propia;
 * desde ahí solo acepta pedidos con esa clave. Para desvincular: Configuración del proyecto → Propiedades → borrar TOKEN.
 *
 * Cada semana es un bloque de 12 filas (como la planilla original):
 *   fila m: mes · m+1: "Lunes DD" ... · m+2: PROPIOS · m+3: USUARIO · m+5: VENTA · m+7: REMANENTE (con el
 *   desposte del día siguiente al lado) · m+9: medias en cámara (y la nota en la columna P).
 * Formato nuevo (desde noviembre 2026): bloque de 13 filas, con "LUNES" (fórmula con el feriado) en m+1 y la FECHA
 *   en m+2; todo lo de abajo queda una fila más abajo (o = 1).
 * Día i (0 = lunes): valor en la columna 3i+2 y, en la fila del remanente, el desposte del día siguiente en 3i+3.
 */
var HOJA = 'Faena-Desposte', ALTO = 12;
var MESES = ['ENERO', 'FEBRERO', 'MARZO', 'ABRIL', 'MAYO', 'JUNIO', 'JULIO', 'AGOSTO', 'SEPTIEMBRE', 'OCTUBRE', 'NOVIEMBRE', 'DICIEMBRE'];
var DIAS = ['Lunes', 'Martes', 'Miercoles', 'Jueves', 'Viernes'];

function doGet(e) { return salida_(manejar_((e && e.parameter) || {}, null)); }
function doPost(e) {
  var b = {};
  try { b = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (x) { return salida_({ ok: false, error: 'pedido ilegible' }); }
  return salida_(manejar_(b, b));
}
function salida_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function manejar_(p, cuerpo) {
  var pr = PropertiesService.getScriptProperties(), tk = pr.getProperty('TOKEN');
  if (!tk) {
    if (p.accion !== 'vincular') return { ok: false, error: 'sin vincular' };
    tk = Utilities.getUuid(); pr.setProperty('TOKEN', tk);
    return { ok: true, token: tk, semanas: leer_(hoja_()) };
  }
  if (p.k !== tk) return { ok: false, error: 'clave incorrecta' };
  var lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    var sh = hoja_();
    if (cuerpo && cuerpo.semanas && cuerpo.semanas.length) escribir_(sh, cuerpo.semanas);
    var est = null; try { est = JSON.parse(pr.getProperty('ESTADO_VENTA') || 'null'); } catch (x) {}
    return { ok: true, semanas: leer_(sh), estadoVenta: est, version: 2 };
  } finally { lock.releaseLock(); }
}
// La hoja Faena-Desposte. Cuando corre por un cambio en la planilla FAENA, "la activa" es FAENA: ahí se abre FAENA POR DÍA por su id.
function hoja_() {
  var pr = PropertiesService.getScriptProperties(), ss = SpreadsheetApp.getActive(), sh = ss && ss.getSheetByName(HOJA);
  if (sh) { if (pr.getProperty('HOJA_ID') !== ss.getId()) pr.setProperty('HOJA_ID', ss.getId()); return sh; }
  var id = pr.getProperty('HOJA_ID');
  if (id && (!ss || ss.getId() !== id)) { var s2 = SpreadsheetApp.openById(id); return s2.getSheetByName(HOJA) || s2.getSheets()[0]; }
  if (ss && ss.getId() === pr.getProperty('FAENA_ID')) throw new Error('Ejecutá traerVenta una vez a mano desde el editor.');
  return ss.getSheets()[0];
}

function norm_(s) { return String(s == null ? '' : s).toUpperCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/\s+/g, ' ').trim(); }
function iso_(d) { return Utilities.formatDate(d, 'GMT', 'yyyy-MM-dd'); }
function num_(v) { if (v === '' || v == null) return null; var n = Number(v); return isNaN(n) ? null : n; }

// Bloques de la hoja: fila del mes (1-based) y el lunes (AAAA-MM-DD). El año es el más cercano a hoy en que esa fecha cae lunes.
function bloques_(vals) {
  var hoy = new Date(), out = [], prev = null, tz = Session.getScriptTimeZone();
  for (var r = 0; r < vals.length; r++) {
    // Formato nuevo: "LUNES" (o "LUNES - FERIADO …") y abajo la fecha del lunes.
    if (/^LUNES( - .*)?$/.test(norm_(vals[r][0])) && r + 1 < vals.length && vals[r + 1][0] instanceof Date) {
      var ln = Utilities.formatDate(vals[r + 1][0], tz, 'yyyy-MM-dd');
      if (prev && ln <= prev) continue;
      out.push({ m: r - 1, lunes: ln, o: 1 }); prev = ln; continue;
    }
    var m = norm_(vals[r][0]).match(/^LUNES\s*(\d{1,2})$/); if (!m) continue;
    var dia = +m[1], mes = r > 0 ? MESES.indexOf(norm_(vals[r - 1][0])) : -1, lun = null;
    if (mes >= 0) {
      var mejor = null;
      for (var y = hoy.getFullYear() - 4; y <= hoy.getFullYear() + 1; y++) {
        var d = new Date(Date.UTC(y, mes, dia));
        if (d.getUTCDay() === 1 && d.getUTCMonth() === mes) { var dif = Math.abs(d - hoy); if (!mejor || dif < mejor.dif) mejor = { d: d, dif: dif }; }
      }
      if (mejor) lun = iso_(mejor.d);
    }
    if (lun && prev && lun <= prev) lun = null;   // bloque repetido (copiado sin cambiar la fecha): va como la semana siguiente
    if (!lun && prev) { var pd = new Date(prev + 'T00:00:00Z'); pd.setUTCDate(pd.getUTCDate() + 7); lun = iso_(pd); }
    if (!lun) continue;
    out.push({ m: r - 1, lunes: lun, o: 0 }); prev = lun;   // m: índice 0-based de la fila del mes (la de arriba de "Lunes")
  }
  return out;
}

function leer_(sh) {
  var rg = sh.getDataRange(), vals = rg.getValues(), fs = rg.getFormulas(), out = [];
  var v = function (r, c) { return r < vals.length && c < vals[r].length ? vals[r][c] : ''; };
  var f = function (r, c) { return r < fs.length && c < fs[r].length ? fs[r][c] : ''; };
  bloques_(vals).forEach(function (b) {
    var m = b.m + (b.o || 0), dias = [];   // con el formato nuevo todo está una fila más abajo
    for (var i = 0; i < 5; i++) {
      var c = 3 * i + 1;   // 0-based: columna del valor del día
      var remF = f(m + 7, c), x = { propios: num_(v(m + 2, c)), usuario: num_(v(m + 3, c)), venta: num_(v(m + 5, c)), despSig: num_(v(m + 7, c + 1)) };
      if (!remF) x.rem = num_(v(m + 7, c));   // remanente escrito a mano (sin fórmula)
      dias.push(x);
    }
    var nota = String(v(m + 9, 15) || '').trim();
    out.push({ lunes: b.lunes, dias: dias, nota: nota });
  });
  return out;
}

// Escribe semanas completas. Si una semana no existe, se agregan bloques al final copiando el último (con su formato y fórmulas).
function escribir_(sh, semanas) {
  semanas = semanas.slice().sort(function (a, b) { return a.lunes < b.lunes ? -1 : 1; });
  semanas.forEach(function (w) {
    var bl = bloques_(sh.getDataRange().getValues()), b = bl.filter(function (x) { return x.lunes === w.lunes; })[0];
    while (!b && bl.length && bl[bl.length - 1].lunes < w.lunes) { b = agregar_(sh, bl[bl.length - 1]); bl.push(b); if (b.lunes !== w.lunes) b = null; }
    if (!b) return;   // semanas anteriores a la primera de la hoja no se agregan
    var m = b.m + 1 + (b.o || 0);  // 1-based (formato nuevo: una fila más abajo)
    var ib = bl.indexOf(b), ant = ib > 0 ? bl[ib - 1] : null;
    (w.dias || []).slice(0, 5).forEach(function (d, i) {
      var c = 3 * i + 2;
      if ('propios' in d) sh.getRange(m + 2, c).setValue(d.propios == null ? '' : d.propios);
      if ('usuario' in d) sh.getRange(m + 3, c).setValue(d.usuario ? d.usuario : '');
      if ('venta' in d) sh.getRange(m + 5, c).setValue(d.venta == null ? '' : d.venta);
      if ('despSig' in d) sh.getRange(m + 7, c + 1).setValue(d.despSig == null ? '' : d.despSig);
      if ('rem' in d) {
        var cel = sh.getRange(m + 7, c);
        if (d.rem != null) cel.setValue(d.rem);
        else if (i > 0) cel.setFormulaR1C1('=R[1]C[-3]-R[0]C[-2]');          // disponible del día anterior − desposte
        else if (ant) cel.setFormula('=P' + (ant.m + 1 + (ant.o || 0) + 7));    // lunes: remanente final de la semana anterior (columna P)
      }
    });
    if ('nota' in w) sh.getRange(m + 9, 16).setValue(w.nota || '');
  });
}
function agregar_(sh, ult) {
  var alto = ALTO + (ult.o || 0), origen = ult.m + 1, destino = origen + alto, falta = destino + alto - 1 - sh.getMaxRows();
  if (falta > 0) sh.insertRowsAfter(sh.getMaxRows(), falta);
  sh.getRange(origen, 1, alto, 18).copyTo(sh.getRange(destino, 1, alto, 18));
  var lun = new Date(ult.lunes + 'T00:00:00Z'); lun.setUTCDate(lun.getUTCDate() + 7);
  sh.getRange(destino, 1).setValue(MESES[lun.getUTCMonth()]);
  if (ult.o) {   // formato nuevo: las fechas de cada día en la fila de abajo de "LUNES" (los nombres y feriados salen solos)
    for (var k = 0; k < 5; k++) { var dk = new Date(lun); dk.setUTCDate(lun.getUTCDate() + k); sh.getRange(destino + 2, 3 * k + 1).setValue(new Date(dk.getUTCFullYear(), dk.getUTCMonth(), dk.getUTCDate())); }
    sh.getRange(destino + 10, 16).setValue('');
    return { m: destino - 1, lunes: iso_(lun), o: 1 };
  }
  for (var i = 0; i < 5; i++) { var d = new Date(lun); d.setUTCDate(lun.getUTCDate() + i); sh.getRange(destino + 1, 3 * i + 1).setValue(DIAS[i] + ' ' + ('0' + d.getUTCDate()).slice(-2)); }
  sh.getRange(destino + 9, 16).setValue('');
  return { m: destino - 1, lunes: iso_(lun) };
}

// ---------- VENTA desde la planilla FAENA (la de clientes por día) ----------
// Menú Qualitá → "Conectar planilla FAENA": se pega el link de la planilla FAENA y queda revisándola cada 10 minutos.
// Por cada día con clientes cargados, el TOTAL de clientes (la fila TOTAL de arriba de DESPOSTE) pasa a VENTA acá,
// y de acá a la pestaña Cámara de la app. Los días sin clientes cargados no se tocan.
function conectarFaena() {
  var ui = SpreadsheetApp.getUi(), r = ui.prompt('Conectar planilla FAENA', 'Pegá el link de la planilla FAENA (la de clientes por día):', ui.ButtonSet.OK_CANCEL);
  if (r.getSelectedButton() !== ui.Button.OK) return;
  var m = String(r.getResponseText()).match(/\/d\/([a-zA-Z0-9_-]{20,})/) || String(r.getResponseText()).trim().match(/^([a-zA-Z0-9_-]{20,})$/);
  if (!m) { ui.alert('No reconozco ese link. Copiá el link completo de la planilla FAENA.'); return; }
  try { SpreadsheetApp.openById(m[1]).getName(); } catch (e) { ui.alert('No pude abrir esa planilla: ' + e.message); return; }
  PropertiesService.getScriptProperties().setProperty('FAENA_ID', m[1]);
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'traerVenta') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('traerVenta').timeBased().everyMinutes(10).create();
  var n = traerVenta();
  ui.alert('Listo: la VENTA de cada día se toma de la planilla FAENA (se revisa cada 10 minutos). Días actualizados ahora: ' + n + '.');
}
// Para que el cambio pase al instante: corre traerVenta cada vez que se edita la planilla FAENA (además del reloj).
function activarAlEditar() {
  var id = PropertiesService.getScriptProperties().getProperty('FAENA_ID'); if (!id) throw new Error('Falta la propiedad FAENA_ID');
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'traerVenta' && t.getEventType() === ScriptApp.EventType.ON_EDIT) ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('traerVenta').forSpreadsheet(id).onEdit().create();
}
function traerVentaAhora() { var n = traerVenta(); SpreadsheetApp.getUi().alert(n == null ? 'Primero conectá la planilla FAENA.' : 'Días actualizados: ' + n + '.'); }

// Lunes de la semana de hoy (sábado y domingo: la que pasó).
function lunesHoy_() {
  var tz = Session.getScriptTimeZone(), ahora = new Date(), dsem = +Utilities.formatDate(ahora, tz, 'u');
  var d = new Date(Utilities.formatDate(ahora, tz, 'yyyy-MM-dd') + 'T00:00:00Z'); d.setUTCDate(d.getUTCDate() - (dsem - 1));
  return iso_(d);
}
// Deja anotado si la venta pasa bien; la app lo muestra con un cartel grande si algo se rompe (por ejemplo, si se cambia el formato).
function estadoVenta_(ok, msg) { PropertiesService.getScriptProperties().setProperty('ESTADO_VENTA', JSON.stringify({ ok: ok, msg: msg || '', ts: new Date().getTime() })); }
var SEM_FAENA = [];   // lunes de las semanas que se encontraron en la planilla FAENA (aunque no tengan clientes)
function traerVenta() {
  var id = PropertiesService.getScriptProperties().getProperty('FAENA_ID'); if (!id) return null;
  SEM_FAENA = [];
  var ventas = ventasFaena_(SpreadsheetApp.openById(id)), n = 0, lh = lunesHoy_();
  var lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    var sh = hoja_(), vals = sh.getDataRange().getValues(), bl = bloques_(vals);
    if (SEM_FAENA.indexOf(lh) < 0) estadoVenta_(false, 'No encuentro la semana del ' + lh.split('-').reverse().slice(0, 2).join('/') + ' en la planilla FAENA. ¿Le cambiaron el formato?');
    else if (!bl.some(function (b) { return b.lunes === lh; })) estadoVenta_(false, 'No encuentro la semana del ' + lh.split('-').reverse().slice(0, 2).join('/') + ' en FAENA POR DÍA. ¿Le cambiaron el formato?');
    else estadoVenta_(true, '');
    bl.forEach(function (b) {
      var lun = new Date(b.lunes + 'T00:00:00Z');
      for (var i = 0; i < 5; i++) {
        var d = new Date(lun); d.setUTCDate(lun.getUTCDate() + i);
        var v = ventas[iso_(d)]; if (v == null) continue;
        var r = b.m + 5 + (b.o || 0), c = 3 * i + 1, ant = r < vals.length ? num_(vals[r][c]) : null;
        if (ant !== v) { sh.getRange(r + 1, c + 1).setValue(v); n++; }
      }
    });
  } finally { lock.releaseLock(); }
  return n;
}

// { 'AAAA-MM-DD': total de clientes } de todas las pestañas "MES AAAA" de la planilla FAENA.
function ventasFaena_(ss) {
  var out = {};
  ss.getSheets().forEach(function (sh) {
    var t = norm_(sh.getName()), mes = -1, y = (t.match(/(20\d\d)/) || [])[1];
    MESES.forEach(function (n, k) { if (t.indexOf(n) === 0) mes = k; });
    if (mes < 0 || !y) return;
    var vals = sh.getDataRange().getValues(), txt = sh.getDataRange().getDisplayValues();
    Object.assign(out, ventasHoja_(vals, txt, +y, mes));
  });
  return out;
}
function ventasHoja_(vals, txt, anio, mes) {
  var out = {}, T = function (r, c) { return r < txt.length && c < txt[r].length ? norm_(txt[r][c]) : ''; };
  for (var r = 0; r < txt.length; r++) {
    if (!/^LUNES\s*\d{1,2}\b/.test(T(r, 1))) continue;   // "Lunes 12" o "Lunes 12 - FERIADO ..."
    // Filas "TOTAL" de la columna B hasta "DESPOSTE": la primera es la de faena y la última la de clientes.
    var tots = [], k;
    for (k = r + 1; k < txt.length && k < r + 60; k++) {
      if (T(k, 1) === 'DESPOSTE' || /^LUNES\s*\d/.test(T(k, 1))) break;
      if (T(k, 1) === 'TOTAL') tots.push(k);
    }
    if (T(k, 1) !== 'DESPOSTE' || tots.length < 2) continue;
    var desde = tots[0] + 1, tot = tots[tots.length - 1];
    for (var i = 0; i < 5; i++) {
      var c = 4 * i + 1, m = T(r, c).match(/^[A-Z]+\s*(\d{1,2})\b/); if (!m) continue;
      var f = fechaDia_(anio, mes, +m[1], i + 1); if (!f) continue;
      if (i === 0 && SEM_FAENA.indexOf(f) < 0) SEM_FAENA.push(f);
      var hay = false;   // ¿hay algún cliente cargado ese día?
      for (var q = desde; q < tot && !hay; q++) hay = !!T(q, c) && num_(vals[q][c + 1]) != null;
      var v = num_(vals[tot][c + 3]); if (v == null) v = num_(vals[tot][c + 1]);
      if (hay && v != null && v > 0) out[f] = v;
    }
  }
  return out;
}
// Fecha del día "dia" (día del mes) que cae en el día de semana dsem (1 = lunes), en el mes de la pestaña o el de al lado.
function fechaDia_(anio, mes, dia, dsem) {
  var mejor = null;
  [-1, 0, 1].forEach(function (k) {
    var d = new Date(Date.UTC(anio, mes + k, dia));
    if (d.getUTCDate() !== dia || d.getUTCDay() !== dsem) return;
    var dif = Math.abs(d - new Date(Date.UTC(anio, mes, 15)));
    if (!mejor || dif < mejor.dif) mejor = { d: d, dif: dif };
  });
  return mejor ? iso_(mejor.d) : null;
}

// Al abrir la planilla, va directo a la semana de hoy (sábado y domingo: a la semana que viene), con el día de hoy marcado.
function onOpen() {
  try {
    SpreadsheetApp.getUi().createMenu('Qualitá').addItem('Conectar planilla FAENA (venta)', 'conectarFaena').addItem('Traer venta de FAENA ahora', 'traerVentaAhora').addToUi();
  } catch (e) {}
  var ss = SpreadsheetApp.getActive(), sh = hoja_(), tz = ss.getSpreadsheetTimeZone() || Session.getScriptTimeZone();
  var ahora = new Date(), dsem = +Utilities.formatDate(ahora, tz, 'u');   // 1 = lunes … 7 = domingo
  var hoy = new Date(Utilities.formatDate(ahora, tz, 'yyyy-MM-dd') + 'T00:00:00Z'), lun = new Date(hoy);
  lun.setUTCDate(hoy.getUTCDate() - (dsem - 1) + (dsem > 5 ? 7 : 0));
  var iso = iso_(lun), bl = bloques_(sh.getDataRange().getValues()), b = null;
  bl.forEach(function (x) { if (x.lunes <= iso) b = x; });
  if (!b) return;
  var fila = b.m + 1, col = dsem > 5 || b.lunes !== iso ? 1 : 3 * (dsem - 1) + 1;
  ss.setActiveSheet(sh);
  sh.setActiveRange(sh.getRange(Math.min(sh.getMaxRows(), fila + 60), 1));   // primero abajo, así al volver la semana queda arriba
  SpreadsheetApp.flush();
  sh.setActiveRange(sh.getRange(fila + 1, col));
}
